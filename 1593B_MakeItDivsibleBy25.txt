import java.util.Scanner;

public class MakeitDivisibleby25 {
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        
        if (!sc.hasNextInt()) return;
        int t = sc.nextInt();
        
        while(t-->0){
            String s = sc.next();
            int minDeletions = Integer.MAX_VALUE;
            
            String[] targets = {"00", "25", "50", "75"};
            
            for(String target : targets){
                int deletions = getDeletions(s, target.charAt(0), target.charAt(1));
                if(deletions != -1){
                    minDeletions = Math.min(minDeletions, deletions);
                }
            }
            System.out.println(minDeletions);
        }
        
        sc.close();
    }
    
    private static int getDeletions(String s, char d1, char d2) {
        int n = s.length();
        int d2Idx = -1;
        for (int i = n - 1; i >= 0; i--){
            if(s.charAt(i) == d2){
                d2Idx = i;
                break;
            }
        }
        
        if(d2Idx == -1) return -1; 
        
        for(int i = d2Idx - 1; i >= 0; i--){
            if (s.charAt(i) == d1){

                return (n - 1 - i) - 1;
            }
        }
        return -1; 
    }
}
