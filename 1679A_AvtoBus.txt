import java.util.Scanner;

public class AvtoBus{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) return;
        int t = sc.nextInt();
        
        while (t-- > 0){
            long n = sc.nextLong();

            if (n % 2 != 0 || n < 4) {
                System.out.println("-1");
            } 
            else {
                long minBuses = (n + 5) / 6;
                long maxBuses = n / 4;
                System.out.println(minBuses + " " + maxBuses);
            }
        }
        
        sc.close();
    }
}
