import java.util.Scanner;

public class StrangePartition{
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) return;

        int t = sc.nextInt();
        while (t-- > 0) {
            int n = sc.nextInt();
            long x = sc.nextLong();

            long totalSum = 0;
            long maxSum = 0;

            for (int i = 0; i < n; i++) {
                long a = sc.nextLong();
                totalSum += a;
                maxSum += (a + x - 1) / x;
            }

            long minSum = (totalSum + x - 1) / x;

            System.out.println(minSum + " " + maxSum);
        }
        sc.close();
    }
}
